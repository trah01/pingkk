cmake_minimum_required(VERSION 3.15)
project(pingkk_portable VERSION 0.1.2 LANGUAGES CXX RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT DEFINED PAYLOAD_DIR)
    message(FATAL_ERROR "PAYLOAD_DIR is required")
endif()
file(TO_CMAKE_PATH "${PAYLOAD_DIR}" PAYLOAD_DIR)
configure_file(../../../src/windows/version.rc.in version.rc @ONLY)
add_executable(pingkk_portable WIN32 launcher.cpp
    "${PAYLOAD_DIR}/payload.rc" "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
target_compile_features(pingkk_portable PRIVATE cxx_std_17)
target_compile_definitions(pingkk_portable PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
    target_compile_definitions(pingkk_portable PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
else()
    target_compile_definitions(pingkk_portable PRIVATE _WIN32_WINNT=0x0601 WINVER=0x0601)
endif()
target_compile_options(pingkk_portable PRIVATE /utf-8 /O1)
target_include_directories(pingkk_portable PRIVATE "${PAYLOAD_DIR}")
target_link_libraries(pingkk_portable PRIVATE setupapi ole32 shell32)
